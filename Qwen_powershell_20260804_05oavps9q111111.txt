# ============================================
# STEP 1: Install AD DS
# ============================================
Write-Step "STEP 1/6 : Install Active Directory Domain Services"

try {
    $adFeature = Get-WindowsFeature AD-Domain-Services -ErrorAction Stop
    if ($adFeature.Installed) {
        Write-OK "AD-Domain-Services already installed"
    } else {
        Write-Info "Installing AD-Domain-Services role..."
        $result = Install-WindowsFeature AD-Domain-Services -IncludeManagementTools
        if ($result.Success) {
            Write-OK "AD-Domain-Services installed successfully"
        } else {
            Write-Fail "Failed to install AD-Domain-Services"; exit 1
        }
    }
} catch {
    Write-Fail "Error checking/installing AD feature: $_"
    exit 1
}

# Check if forest exists
try {
    $domain = Get-ADDomain -ErrorAction Stop
    Write-OK "Forest already exists: $($domain.DNSRoot)"
} catch {
    Write-Info "No forest found. Creating new forest: $DomainName ..."
    try {
        $securePass = ConvertTo-SecureString $SafeModePassword -AsPlainText -Force
        Install-ADDSForest `
            -DomainName $DomainName `
            -DomainNetbiosName $DomainNetBIOS `
            -SafeModeAdministratorPassword $securePass `
            -InstallDns:$true `
            -NoRebootOnCompletion:$false `
            -Force `
            -ErrorAction Stop | Out-Null
        
        Write-Warn2 "Server is rebooting now..."
        Write-Warn2 "After reboot, run this script AGAIN to continue setup."
        Write-Host ""
        Read-Host "Press Enter to exit"
        exit 0
    } catch {
        Write-Fail "Failed to create forest: $_"
        exit 1
    }
}

# Wait for AD to be ready
Write-Info "Waiting for AD to be ready..."
$retries = 0
$maxRetries = 60
while ($retries -lt $maxRetries) {
    try {
        $null = Get-ADDomain -ErrorAction Stop
        Write-OK "AD is ready (waited $($retries * 3) seconds)"
        break
    } catch {
        Start-Sleep -Seconds 3
        $retries++
        Write-Host "." -NoNewline -ForegroundColor DarkGray
    }
}
if ($retries -ge $maxRetries) {
    Write-Fail "AD did not become ready after 180 seconds"
    exit 1
}

# Create OUs

foreach ($ouName in $ouList) {
    try {
        $existing = Get-ADOrganizationalUnit -Filter "Name -eq '$ouName'" -ErrorAction SilentlyContinue

            New-ADOrganizationalUnit -Name $ouName -ProtectedFromAccidentalDeletion $false -ErrorAction Stop | Out-Null
            Write-OK "OU created: $ouName"
        } else {
            Write-Info "OU already exists: $ouName"

        Write-Fail "Failed to create OU $ouName : $_"
