function Write-Step {
    param([string]$msg, [string]$color = "Cyan")
    Write-Host ""
    Write-Host "========================================" -ForegroundColor DarkGray
    Write-Host ("  " + $msg) -ForegroundColor $color
    Write-Host "========================================" -ForegroundColor DarkGray
}

function Write-OK {
    param([string]$msg)
    Write-Host ("  [OK] " + $msg) -ForegroundColor Green
}

function Write-Warn2 {
    param([string]$msg)
    Write-Host ("  [!!] " + $msg) -ForegroundColor Yellow
}

function Write-Fail {
    param([string]$msg)
    Write-Host ("  [FAIL] " + $msg) -ForegroundColor Red
}

function Write-Info {
    param([string]$msg)
    Write-Host ("  [..] " + $msg) -ForegroundColor Gray
}

# ============================================
# STEP 1: Install AD DS
# ============================================
Write-Step -msg "STEP 1/6 : Install Active Directory Domain Services"

try {
    $adFeature = Get-WindowsFeature AD-Domain-Services -ErrorAction Stop
    if ($adFeature.Installed) {
        Write-OK -msg "AD-Domain-Services already installed"
    }
    else {
        Write-Info -msg "Installing AD-Domain-Services role..."
        $result = Install-WindowsFeature AD-Domain-Services -IncludeManagementTools
        if ($result.Success) {
            Write-OK -msg "AD-Domain-Services installed successfully"
        }
        else {
            Write-Fail -msg "Failed to install AD-Domain-Services"
            exit 1
        }
    }
}
catch {
    Write-Fail -msg ("Error checking/installing AD feature: " + $_.Exception.Message)
    exit 1
}

# Check if forest exists
try {
    $domain = Get-ADDomain -ErrorAction Stop
    Write-OK -msg ("Forest already exists: " + $domain.DNSRoot)
}
catch {
    Write-Info -msg ("No forest found. Creating new forest: " + $DomainName + " ...")
    try {
        $securePass = ConvertTo-SecureString $SafeModePassword -AsPlainText -Force
        Install-ADDSForest `
            -DomainName $DomainName `
            -DomainNetbiosName $DomainNetBIOS `
            -SafeModeAdministratorPassword $securePass `
            -InstallDns:$true `
            -NoRebootOnCompletion:$false `
            -Force `
            -ErrorAction Stop | Out-Null

        Write-Warn2 -msg "Server is rebooting now..."
        Write-Warn2 -msg "After reboot, run this script AGAIN to continue setup."
        Write-Host ""
        Read-Host "Press Enter to exit"
        exit 0
    }
    catch {
        Write-Fail -msg ("Failed to create forest: " + $_.Exception.Message)
        exit 1
    }
}

        break
    }
    catch {
        Start-Sleep -Seconds 3
        $retries++
        Write-Host "." -NoNewline -ForegroundColor DarkGray

    Write-Fail -msg "AD did not become ready after 180 seconds"

# Create OUs

foreach ($ouName in $ouList) {
    try {
        $existing = Get-ADOrganizationalUnit -Filter ("Name -eq '" + $ouName + "'") -ErrorAction SilentlyContinue
        if (-not $existing) {
            New-ADOrganizationalUnit -Name $ouName -ProtectedFromAccidentalDeletion $false -ErrorAction Stop | Out-Null
            Write-OK -msg ("OU created: " + $ouName)
        }
        else {
            Write-Info -msg ("OU already exists: " + $ouName)
        }
    }
    catch {
        Write-Fail -msg ("Failed to create OU " + $ouName + " : " + $_.Exception.Message)
    }
